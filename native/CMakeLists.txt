cmake_minimum_required(VERSION 3.22)
project(fgtox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID)
    set(TOX_DIR "${TOX_ROOT}/toxcore-android-${ANDROID_ABI}")
else()
    find_package(JNI REQUIRED)
    set(TOX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.native-deps/toxcore-linux-x86_64")
endif()
add_library(toxcore SHARED IMPORTED)
set_target_properties(toxcore PROPERTIES IMPORTED_LOCATION "${TOX_DIR}/lib/libtoxcore.so")
add_library(fgtox SHARED fgtox.cpp)
target_include_directories(fgtox PRIVATE "${TOX_DIR}/include" ${JNI_INCLUDE_DIRS})
target_link_libraries(fgtox PRIVATE toxcore)
target_compile_options(fgtox PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    target_link_options(fgtox PRIVATE "-Wl,-z,max-page-size=16384")
endif()
